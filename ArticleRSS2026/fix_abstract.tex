\usepackage{lmodern}   % scalable fonts: text and math track \normalsize
\newcommand{\rc}[1]{r_{\scriptscriptstyle #1}}
\newcommand{\Rc}[1]{R_{\scriptscriptstyle #1}}
\makeatletter
% RSS submission requirement: 12 pt throughout.
% statsoc.cls hardcodes normalsize to 11 pt; override it here,
% after the class has finished loading.
\renewcommand\normalsize{%
   \@setfontsize\normalsize{12}{14.5}%
   \abovedisplayskip 10\p@ \@plus2\p@ \@minus5\p@
   \abovedisplayshortskip \z@ \@plus3\p@
   \belowdisplayshortskip 6\p@ \@plus3\p@ \@minus3\p@
   \belowdisplayskip \abovedisplayskip
   \let\@listi\@listI}
\normalsize  % activate immediately so the whole document uses 12 pt
% Abstract: now also 12 pt (normalsize), sans-serif for visual
% distinction, bold roman "Summary" label.
\renewenvironment{abstract}{%
  \list{}{\leftmargin=2pc\parsep\z@\itemsep 3\p@}%
  \item[]\relax
  \sffamily\normalsize\textbf{\abstractname}.\enspace}%
  {\endlist}
% Fix keywords: rticles calls \keywords{} after \end{abstract}, so the
% class's storage-based mechanism never fires. Output immediately.
\def\keywords#1{\medskip\par\noindent\sffamily\textsl{Keywords}: #1\par}
\makeatother
